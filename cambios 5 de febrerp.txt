import requests
import sys

# ===============================
# 🔹 AÑADIDO
# Para trabajar con archivos locales (cache)
# ===============================
import os
import json

# ===============================
# 🔹 AÑADIDO
# Importamos la API key y la URL base
# desde el archivo api.py
# ===============================
from api import API_KEY, BASE_URL

# ===============================
# 🔹 AÑADIDO
# Archivo local donde se guardan los municipios
# ===============================
MUNICIPIOS_FILE = "municipios.json"


# ===============================
# FUNCIONES
# ===============================
def llamada_aemet(url):
    """
    Llamada estándar a la API de AEMET:
    1) llamada al endpoint
    2) llamada a la URL temporal 'datos'
    """
    r = requests.get(url)
    if r.status_code != 200:
        print("❌ Error HTTP:", r.status_code)
        return None

    respuesta = r.json()

    # ===============================
    # 🔹 AÑADIDO
    # AEMET devuelve estado 404 cuando
    # no hay datos disponibles (no es error técnico)
    # ===============================
    if respuesta.get("estado") == 404:
        return None

    if "datos" not in respuesta:
        print("❌ Respuesta inesperada:", respuesta)
        return None

    # Segunda llamada REAL (obligatoria en AEMET)
    datos_url = respuesta["datos"]
    r2 = requests.get(datos_url)

    if r2.status_code != 200:
        return None

    return r2.json()


def cargar_municipios():
    """
    ===============================
    🔹 FUNCIÓN NUEVA
    - Si existe municipios.json → se usa
    - Si no existe → se descarga de AEMET
    ===============================
    """
    if os.path.exists(MUNICIPIOS_FILE):
        with open(MUNICIPIOS_FILE, "r", encoding="utf-8") as f:
            return json.load(f)

    print("📡 Descargando municipios desde AEMET...")
    url = f"{BASE_URL}/maestro/municipios?api_key={API_KEY}"
    municipios = llamada_aemet(url)

    if not municipios:
        print("❌ No se pudieron cargar los municipios")
        sys.exit(1)

    # ===============================
    # 🔹 AÑADIDO
    # Guardamos los municipios en local
    # ===============================
    with open(MUNICIPIOS_FILE, "w", encoding="utf-8") as f:
        json.dump(municipios, f, ensure_ascii=False, indent=2)

    print("💾 Municipios guardados en local")
    return municipios


def obtener_codigo_municipio(nombre):
    """
    Busca el código del municipio por nombre
    """
    # ===============================
    # 🔹 CAMBIO
    # Ahora se leen los municipios desde cache local
    # ===============================
    municipios = cargar_municipios()
    nombre = nombre.lower()

    for m in municipios:
        if m["nombre"].lower() == nombre:
            return m["id"][2:]  # quitamos el prefijo "id"

    return None


def prediccion_horaria(codigo):
    """
    Muestra la predicción horaria:
    temperatura y estado del cielo
    """
    url = f"{BASE_URL}/prediccion/especifica/municipio/horaria/{codigo}?api_key={API_KEY}"
    datos = llamada_aemet(url)

    # ===============================
    # 🔹 AÑADIDO
    # Evita que el programa se rompa si no hay datos
    # ===============================
    if not datos:
        print("⚠️ No hay predicción horaria disponible para este municipio")
        return

    dias = datos[0]["prediccion"]["dia"]

    for dia in dias:
        print(f"\n📅 {dia['fecha']}")

        for t in dia["temperatura"]:
            hora = t["periodo"]
            temp = t["value"]

            estado = next(
                (e["descripcion"] for e in dia["estadoCielo"] if e["periodo"] == hora),
                "N/A"
            )

            print(f"{hora}:00 → {temp} ºC | {estado}")

municipio = input("Introduce el municipio (ejemplo: Montehermoso): ").strip()

codigo = obtener_codigo_municipio(municipio)

if not codigo:
    print("❌ Municipio no encontrado")
    sys.exit(1)

print(f"✅ Municipio encontrado. Código: {codigo}")
prediccion_horaria(codigo)
